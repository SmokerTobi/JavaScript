# Experiment/ Case Study No.: 9

## 1. Experiment Title:
Theme Preferences using `localStorage` (Persistent User Preferences with Light, Dark, and Clear Themes)

## 2. Software/Tools Required:
* **Operating System:** Windows 11
* **Development Environment:** VS Code
* **Languages:** HTML5, CSS3, JavaScript (ES6)
* **Execution Environment:** Web Browser (Google Chrome / Microsoft Edge / Mozilla Firefox)

## 3. Experiment Program Code:

### A. HTML/JS File (`themepref.html`)
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Theme Preferences</title>
    <style>
        /* Most Basic CSS */
        body {
            font-family: sans-serif;
            margin: 50px;
            transition: background-color 0.3s, color 0.3s;
        }

        /* Light Theme */
        body.theme-light {
            background-color: #ffffff;
            color: #000000;
        }

        /* Dark Theme */
        body.theme-dark {
            background-color: #222222;
            color: #ffffff;
        }

        /* Clear Theme */
        body.theme-clear {
            background-color: transparent;
            color: inherit;
        }

        .container {
            border: 1px solid #cccccc;
            padding: 20px;
            max-width: 600px;
        }

        button {
            padding: 10px 15px;
            margin-right: 10px;
            font-size: 16px;
            cursor: pointer;
        }
    </style>
</head>
<body class="theme-light">

    <div class="container">
        <h1>Theme Preferences</h1>
        <p>Choose your preferred theme:</p>
        
        <button onclick="setTheme('theme-light')">Light Theme</button>
        <button onclick="setTheme('theme-dark')">Dark Theme</button>
        <button onclick="setTheme('theme-clear')">Clear Theme</button>
    </div>

    <script>
        function setTheme(themeName) {
            // Remove all theme classes
            document.body.classList.remove('theme-light', 'theme-dark', 'theme-clear');
            
            // Add the selected theme class
            document.body.classList.add(themeName);
            
            // Save the preference
            localStorage.setItem('user-theme', themeName);
        }

        // On page load, apply the saved theme if it exists
        window.onload = function() {
            var savedTheme = localStorage.getItem('user-theme');
            if (savedTheme) {
                setTheme(savedTheme);
            }
        };
    </script>

</body>
</html>
```

### C. JavaScript File (`casestudy/script.js`)
```javascript
document.addEventListener('DOMContentLoaded', () => {
    // DOM Elements
    const inputs = document.querySelectorAll('.topic-input');
    const btnSaveLocal = document.getElementById('btn-save-local');
    const btnSaveSession = document.getElementById('btn-save-session');
    const btnClear = document.getElementById('btn-clear');
    const alertBtns = document.querySelectorAll('.alert-btn');

    // Storage Keys
    const LOCAL_KEY = 'seminar_schedule_local';
    const SESSION_KEY = 'seminar_schedule_session';

    // Helper: Get current data from all inputs
    function getScheduleData() {
        const data = {};
        inputs.forEach(input => {
            data[input.id] = input.value;
        });
        return data;
    }

    // Helper: Populate inputs from a data object
    function populateInputs(data) {
        if (!data) return;
        inputs.forEach(input => {
            if (data[input.id] !== undefined) {
                input.value = data[input.id];
            }
        });
    }

    // Load initial data
    function loadData() {
        const sessionData = sessionStorage.getItem(SESSION_KEY);
        const localData = localStorage.getItem(LOCAL_KEY);

        // Prioritize session storage over local storage
        if (sessionData) {
            populateInputs(JSON.parse(sessionData));
            console.log("Loaded from Session Storage");
        } else if (localData) {
            populateInputs(JSON.parse(localData));
            console.log("Loaded from Local Storage");
        }
    }

    // Event: Save to Local Storage
    btnSaveLocal.addEventListener('click', () => {
        const data = getScheduleData();
        localStorage.setItem(LOCAL_KEY, JSON.stringify(data));
        alert("Schedule saved to Local Storage successfully.");
    });

    // Event: Save to Session Storage
    btnSaveSession.addEventListener('click', () => {
        const data = getScheduleData();
        sessionStorage.setItem(SESSION_KEY, JSON.stringify(data));
        alert("Schedule saved to Session Storage successfully. (Will reset if tab is closed)");
    });

    // Event: Clear Schedule
    btnClear.addEventListener('click', () => {
        // Clear DOM
        inputs.forEach(input => input.value = '');
        
        // Clear Storage
        localStorage.removeItem(LOCAL_KEY);
        sessionStorage.removeItem(SESSION_KEY);
        
        alert("Schedule and storage cleared.");
    });

    // Event: Click on topic cell (via the alert button)
    alertBtns.forEach(btn => {
        btn.addEventListener('click', (e) => {
            // Find the sibling input in the same wrapper
            const wrapper = e.target.closest('.topic-wrapper');
            const input = wrapper.querySelector('.topic-input');
            const topicName = input.value.trim();
            
            if (topicName) {
                alert(`You clicked: ${topicName}`);
            } else {
                alert("You clicked an empty topic cell. Please enter a topic first.");
            }
        });
    });

    // Initialize
    loadData();
});
```

### Expected Output Description:
1. A styled **Technical Seminar Schedule** table is displayed with days (Monday, Tuesday, Wednesday), time slots, and editable topic input fields.
2. User enters topics in the input fields (e.g., Monday: `"Introduction to AI"`, `"Machine Learning Basics"`).
3. Clicking **Save to Local Storage** persists the schedule data permanently (survives browser close). An alert confirms: `"Schedule saved to Local Storage successfully."`.
4. Clicking **Save to Session Storage** persists data for the current session only (cleared when tab is closed). An alert confirms: `"Schedule saved to Session Storage successfully. (Will reset if tab is closed)"`.
5. Clicking the 📌 button next to any topic input shows an alert with the entered topic name (e.g., `"You clicked: Introduction to AI"`).
6. Clicking **Clear** removes all topic data from inputs and both storage mechanisms.
7. On page reload, saved data is automatically restored (session storage is prioritized over local storage).

## 8. Result/Conclusion:
Successfully implemented theme preference persistence using `localStorage` to save and restore user-selected themes (Light, Dark, Clear) across browser sessions. The Case Study demonstrated a Technical Seminar Schedule Manager utilizing both `localStorage` (persistent) and `sessionStorage` (session-scoped) for data management, along with `JSON.stringify`/`JSON.parse` for serializing complex data objects, and interactive `click` event handlers for topic display alerts.
