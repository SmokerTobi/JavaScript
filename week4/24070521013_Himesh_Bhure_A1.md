```html
        }

            margin-top: 20px;
            font-size: 18px;

        }
    </style>
</head>
<body>
    <div class="container">
        <h1>Palindrome Checker</h1>
        <input type="text" id="text-input" placeholder="Enter text">
        <button id="check-btn">Check</button>
        <div id="result"></div>
    </div>

    <script>
        const btn = document.getElementById("check-btn");
        const input = document.getElementById("text-input");
        const result = document.getElementById("result");

        btn.addEventListener("click", () => {
            const value = input.value;
            if (!value) {
                alert("Please input a value");
                return;
            }
            
            // Remove non-alphanumeric characters and make lowercase
            const cleanStr = value.replace(/[^A-Za-z0-9]/g, '').toLowerCase();
            const reversedStr = cleanStr.split('').reverse().join('');
            
            if (cleanStr === reversedStr) {
                result.innerHTML = `${value} is a palindrome<br><br><span style="font-size: 14px; font-weight: normal; color: #666;">Done by Himesh</span>`;
            } else {
                result.innerHTML = `${value} is not a palindrome<br><br><span style="font-size: 14px; font-weight: normal; color: #666;">Done by Himesh</span>`;
            }
        });
    </script>
</body>
</html>
```

## 5. Case Study Title:
Indian Vehicle Registration Validator with Regular Expressions (Regex) and Exception Handling

## 6. Case Study Program Code:

### A. HTML/JS File (`vehicle_registration.html`)
```html
<!DOCTYPE html>
<html>
<head>
    <title>Vehicle Registration Validation</title>
    <style>
        body {
            font-family: Arial, sans-serif;
            margin: 50px;
            background-color: #eafaf1; /* light green background */
            color: #0b5345; /* dark green text */
        }
        .container {
            padding: 40px;
            background-color: #ffffff;
            width: 450px;
            margin: 0 auto;
            text-align: center;
            border-radius: 8px;
            box-shadow: 0 4px 12px rgba(39, 174, 96, 0.15);
        }
        input {
            padding: 10px;
            font-size: 16px;
            width: 80%;
            border: 1px solid #2ecc71;
            border-radius: 4px;
            margin-bottom: 15px;
            box-sizing: border-box;
        }
        input:focus {
            outline: none;
            border-color: #27ae60;
            box-shadow: 0 0 5px rgba(39, 174, 96, 0.3);
        }
        button {
            padding: 10px 24px;
            font-size: 16px;
            background-color: #27ae60;
            color: white;
            border: none;
            border-radius: 4px;
            cursor: pointer;
            transition: background-color 0.2s;
        }
        button:hover {
            background-color: #219150;
        }
        #result {
            margin-top: 20px;
            font-size: 16px;
            font-weight: bold;
        }
        .valid {
            color: #27ae60;
        }
        .invalid {
            color: #e74c3c;
        }
    </style>
</head>
<body>
    <div class="container">
        <h2>Vehicle Registration Validator</h2>
        <input type="text" id="reg-input" placeholder="e.g. MH12AB1234">
        <br>
        <button id="validate-btn">Validate</button>
        <div id="result"></div>
    </div>

    <script>
        function validateRegistration(regNo) {
            if (!regNo) {
                throw new Error("Registration number should not be empty.");
            }
            if (regNo.length !== 10) {
                throw new Error("Length should be exactly 10 characters.");
            }
            // 1. First two characters must be uppercase alphabets (State Code)
            if (!/^[A-Z]{2}$/.test(regNo.substring(0, 2))) {
                throw new Error("First two characters must be uppercase alphabets.");
            }
            // 2. Next two characters must be digits (District Code)
            if (!/^\d{2}$/.test(regNo.substring(2, 4))) {
                throw new Error("Third and fourth characters must be digits.");
            }
            // 3. Next two characters must be uppercase alphabets (Series)
            if (!/^[A-Z]{2}$/.test(regNo.substring(4, 6))) {
                throw new Error("Fifth and sixth characters must be uppercase alphabets.");
            }
            // 4. Last four characters must be digits (Vehicle Number)
            if (!/^\d{4}$/.test(regNo.substring(6, 10))) {
                throw new Error("Last four characters must be digits.");
            }
            
            return "Valid Vehicle Registration Number!";
        }

        document.getElementById("validate-btn").addEventListener("click", () => {
            const input = document.getElementById("reg-input").value;
            const resultDiv = document.getElementById("result");
            
            try {
                const message = validateRegistration(input);
                resultDiv.textContent = message;
                resultDiv.className = "valid";
            } catch (error) {
                resultDiv.textContent = "Invalid: " + error.message;
                resultDiv.className = "invalid";
            }
        });
    </script>
</body>
</html>
```
