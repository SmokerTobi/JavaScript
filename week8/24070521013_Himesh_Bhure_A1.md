# Experiment/ Case Study No.: 8

## 1. Experiment Title:
Gym Admission Form with Real-Time Input Validation using JavaScript Events (`input`, `blur`, `change`, `submit`)

## 2. Software/Tools Required:
* **Operating System:** Windows 11
* **Development Environment:** VS Code
* **Languages:** HTML5, CSS3, JavaScript (ES6)
* **Execution Environment:** Web Browser (Google Chrome / Microsoft Edge / Mozilla Firefox)

## 3. Experiment Program Code:

### A. HTML/JS File (`validation.html`)
```html
<!DOCTYPE html>
<head>
    <title>Gym Admmission Form</title>
    <style>
        body{
            font-family: Arial, sans-serif;
            background : #f2f2f2;
        }

        .container{
            width : 400px;
            margin : 40px auto;
            background : white;
            padding : 20px;
            border-radius : 10px;
            box-shadow : 0 0 10px gray;
        }

        h2{
            text-align: center;
            
        }

        label{
            display : block;
            margin-top: 15px;
        }

        input,select{
            width : 100%;
            padding: 8px;
            margin-top: 5px;
        }
        .error{
            color : red;
            font-size : 13px;
        }

        .success{
            color : green;
            font-size : 14px;
            text-align: center;
        }
        button{
            width : 100%;
            padding : 10px;
            margin-top : 15px;
            background: green;
            color : white;
            border : none;
            cursor : pointer;
        }
        
    </style>
</head>

<body>
    <div class="container">
        <h2>Gym Admmission Form</h2>

        <form id="gymForm">
            <label>Full Name</label>
            <input type="text" id = "name">
            <span id ="nameError" class="error"></span>

            <label>Email</label>
            <input type="email" id = "email">
            <span id ="emailError" class="error"></span>

            <label>Mobile Number</label>
            <input type="number" id = "mobile">
            <span id ="mobileError" class="error"></span>

            <label>Age</label>
            <input type="number" id = "age">
            <span id ="ageError" class="error"></span>

            <label>Membership Plan</label>
            <select id ="plan">
                <option value="">Select Plan</option>
                <option value="monthly">Monthly</option>
                <option value="quarterly">Quarterly</option>
                <option value="yearly">Yearly</option>
            </select>
            <span id ="planError" class="error"></span>

            <button type="submit">Submit</button>
            <p id ="result" class="success"></p>
        </form>
    </div>
    <script>
        //Name Validation (input event)
        document.getElementById("name").addEventListener("input", function(){
            let name = this.value;
            if(/^[A-Za-z ]+$/.test(name)){
                document.getElementById("nameError").innerHTML = "";
            }else{
                document.getElementById("nameError").innerHTML = "Only letters and spaces allowed";
            }      
        });

        //Age Validation (blur event)
        document.getElementById("age").addEventListener("blur", function(){
            let age = this.value;
            if(age >= 16 && age <= 60){
                document.getElementById("ageError").innerHTML = "";
            }else{
                document.getElementById("ageError").innerHTML = "Age must be between 16 and 60";
            }      
        });

        //Email Validation (input)
        document.getElementById("email").addEventListener("input",function(){
            let email = this.value;
            let pattern = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
            if (pattern.test(email)) {
                document.getElementById("emailError").innerHTML = "";
            } else {
                document.getElementById("emailError").innerHTML = "Invalid Email Address";
            }
        });

        //Mobile Validation
        document.getElementById("mobile").addEventListener("input",function(){
            let mobile = this.value;
            if(/^\d{10}$/.test(mobile)){
                document.getElementById("mobileError").innerHTML = "";
            }else{
                document.getElementById("mobileError").innerHTML = "Invalid Mobile Number";
            }    
        })

        //Membership Validation
        document.getElementById("plan").addEventListener("change", function(){
            let plan = this.value;
            if(plan != ""){
                document.getElementById("planError").innerHTML = "";
            }else{
                document.getElementById("planError").innerHTML = "Select a Membership Plan";
            }
        });

        //Form Submission
        document.getElementById("gymForm").addEventListener("submit", function(e){
            e.preventDefault();
            
            if(
                document.getElementById("nameError").innerHTML == "" &&
                document.getElementById("emailError").innerHTML == "" &&
                document.getElementById("mobileError").innerHTML == "" &&
                document.getElementById("ageError").innerHTML == "" &&
                document.getElementById("planError").innerHTML == ""
            ){
                document.getElementById("result").innerHTML = "Gym Admission Successful!";
            }else{
                document.getElementById("result").innerHTML = "Please Fill All Fields Correctly.";
            }
        });
    </script>
</body>
</html>
```

## 4. Output:
*(Attach screenshot/output, /PRN, Name and File Path with your name is mandatory in screenshot)*

### Expected Output Description:
1. A Gym Admission Form is displayed with fields: Full Name, Email, Mobile Number, Age, and Membership Plan.
2. **Name Validation** (`input` event): Typing non-alphabetic characters shows `"Only letters and spaces allowed"` in red.
3. **Age Validation** (`blur` event): Entering an age outside 16-60 range and clicking away shows `"Age must be between 16 and 60"`.
4. **Email Validation** (`input` event): Typing an invalid email shows `"Invalid Email Address"`.
5. **Mobile Validation** (`input` event): Entering a number that isn't exactly 10 digits shows `"Invalid Mobile Number"`.
6. **Plan Validation** (`change` event): Not selecting a plan shows `"Select a Membership Plan"`.
7. On clicking **Submit** with all valid fields, the success message `"Gym Admission Successful!"` appears in green.

## 5. Case Study Title:
Enhanced Gym Admission Registration Form with Comprehensive Event-Based Validation and Styled UI

## 6. Case Study Program Code:

### A. HTML/CSS/JS File (`case_study.html`)
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Gym Admission Registration</title>
    <style>
        :root {
            --primary: #4F46E5;
            --primary-hover: #4338CA;
            --surface: #ffffff;
            --background: #F3F4F6;
            --text-dark: #111827;

            --error: #DC2626;
            --success: #059669;
            --success-bg: #D1FAE5;
            --error-bg: #FEE2E2;
        }

        body {
            font-family: -apple-system, BlinkMacSystemFont, "Segoe UI", Roboto, Helvetica, Arial, sans-serif;
            background-color: var(--background);
            color: var(--text-dark);
            margin: 0;
            padding: 40px 20px;
            display: flex;
            justify-content: center;
        }

        .form-container {
            background-color: var(--surface);
            max-width: 700px;
            width: 100%;
            border-radius: 8px;
            box-shadow: 0 4px 6px -1px rgba(0, 0, 0, 0.1), 0 2px 4px -1px rgba(0, 0, 0, 0.06);
            overflow: hidden;
        }

        .form-header {
            background-color: var(--primary);

            color: white;
        }

        .form-header h2 { margin: 0; font-size: 1.5rem; font-weight: 600; }
        .form-header p { margin: 8px 0 0 0; color: #E0E7FF; font-size: 0.95rem; }

        form { padding: 32px; }

        .form-group {
            display: flex; flex-direction: row;
            margin-bottom: 24px; align-items: flex-start;
        }

        .form-label {
            width: 200px; font-weight: 500; color: var(--text-gray);

        }

        .form-input-container { flex-grow: 1; position: relative; }

        input[type="text"], input[type="email"], input[type="number"], select, textarea {
            width: 100%; padding: 10px 14px; border: 1px solid var(--border);
            border-radius: 6px; font-size: 1rem; color: var(--text-dark);
            background-color: #fff; transition: border-color 0.15s ease-in-out;
            box-sizing: border-box;
        }

        input:focus, select:focus, textarea:focus {
            outline: none; border-color: var(--primary);
            box-shadow: 0 0 0 3px rgba(79, 70, 229, 0.15);
        }

        textarea { resize: vertical; min-height: 80px; }

        .options-group { display: flex; gap: 24px; padding-top: 10px; flex-wrap: wrap; }
        .option-item { display: flex; align-items: center; gap: 8px; cursor: pointer; }

        input[type="radio"], input[type="checkbox"] {
            width: 16px; height: 16px; accent-color: var(--primary); cursor: pointer; margin: 0;
        }

        .error-message { color: var(--error); font-size: 0.85rem; margin-top: 6px; display: none; }

        .submit-btn {
            background-color: var(--primary); color: white; border: none;
            padding: 12px 24px; font-size: 1rem; font-weight: 500;
            border-radius: 6px; cursor: pointer; width: 100%;
            transition: background-color 0.15s ease-in-out; margin-top: 12px;
        }

        }

    <div class="form-container">
        <div class="form-header">
            <h2>Gym Admission Form</h2>
            <p>Please enter your details to register</p>
        </div>

        <form id="registrationForm">
            <div class="form-group">
                <div class="form-label">Name</div>
                <div class="form-input-container">
                    <input type="text" id="name" placeholder="Full Name">
                    <div id="nameError" class="error-message"></div>
                </div>
            </div>

            <div class="form-group">
                <div class="form-label">Email</div>
                <div class="form-input-container">
                    <input type="email" id="email" placeholder="Email Address">
                    <div id="emailError" class="error-message"></div>
                </div>
            </div>

            <div class="form-group">
                <div class="form-label">Mobile Number</div>
                <div class="form-input-container">
                    <input type="number" id="mobile" placeholder="10-digit Number">
                    <div id="mobileError" class="error-message"></div>
                </div>
            </div>

            <div class="form-group">
                <div class="form-label">Age</div>
                <div class="form-input-container">
                    <input type="number" id="age" placeholder="Between 16 and 60">
                    <div id="ageError" class="error-message"></div>
                </div>
            </div>

            <div class="form-group">
                <div class="form-label">Sex</div>
                <div class="form-input-container">
                    <div class="options-group">
                        <label class="option-item"><input type="radio" name="sex" value="Male" checked> Male</label>
                        <label class="option-item"><input type="radio" name="sex" value="Female"> Female</label>
                    </div>
                </div>
            </div>

            <div class="form-group">
                <div class="form-label">Height</div>
                <div class="form-input-container">
                    <input type="text" id="height" placeholder="e.g., 5'10&quot; or 178cm">
                </div>
            </div>

            <div class="form-group">
                <div class="form-label">BMI</div>
                <div class="form-input-container">
                    <input type="number" id="bmi" step="0.1" placeholder="e.g., 22.5">
                </div>
            </div>

            <div class="form-group">
                <div class="form-label">Membership Plan</div>
                <div class="form-input-container">
                    <select id="plan">
                        <option value="">Select a Plan...</option>
                        <option value="monthly">Monthly</option>
                        <option value="quarterly">Quarterly</option>
                        <option value="yearly">Yearly</option>
                    </select>
                    <div id="planError" class="error-message"></div>
                </div>
            </div>

            <div class="form-group">
                <div class="form-label">Describe your athletic ability</div>
                <div class="form-input-container">
                    <textarea id="ability" placeholder="Briefly describe..."></textarea>
                </div>
            </div>

            <button type="submit" class="submit-btn">Enter my information</button>
            <div id="result" class="result-message"></div>
        </form>
    </div>

    <script>
        const form = document.getElementById('registrationForm');
        
        const inputs = {
            name: document.getElementById('name'),
            email: document.getElementById('email'),
            mobile: document.getElementById('mobile'),
            age: document.getElementById('age'),
            plan: document.getElementById('plan')
        };

        const showError = (elementId, message) => {
            const errorEl = document.getElementById(elementId);
            errorEl.textContent = message;
            errorEl.style.display = message ? 'block' : 'none';
        };

        const validateName = () => {
            const val = inputs.name.value;
            if(!val) return false;
            if(/^[A-Za-z ]+$/.test(val)){
                showError('nameError', ''); return true;
            } else {
                showError('nameError', 'Only letters and spaces allowed'); return false;
            }
        };

        const validateEmail = () => {
            const val = inputs.email.value;
            if(!val) return false;
            let pattern = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
            if (pattern.test(val)) {
                showError('emailError', ''); return true;
            } else {
                showError('emailError', 'Invalid Email Address'); return false;
            }
        };

        const validateMobile = () => {
            const val = inputs.mobile.value;
            if(!val) return false;
            if(/^\d{10}$/.test(val)){
                showError('mobileError', ''); return true;
            } else {
                showError('mobileError', 'Invalid Mobile Number (10 digits)'); return false;
            }
        };

        const validateAge = () => {
            const val = inputs.age.value;
            if(!val) return false;
            if(val >= 16 && val <= 60){
                showError('ageError', ''); return true;
            } else {
                showError('ageError', 'Age must be between 16 and 60'); return false;
            }
        };

        const validatePlan = () => {
            if(inputs.plan.value !== ""){
                showError('planError', ''); return true;
            } else {
                showError('planError', 'Select a Membership Plan'); return false;
            }
        };

        inputs.name.addEventListener('input', validateName);
        inputs.email.addEventListener('input', validateEmail);
        inputs.mobile.addEventListener('input', validateMobile);
        inputs.age.addEventListener('blur', validateAge);
        inputs.plan.addEventListener('change', validatePlan);

        form.addEventListener('submit', function(e){
            e.preventDefault();
            
            const isNameValid = validateName();
            const isEmailValid = validateEmail();
            const isMobileValid = validateMobile();
            const isAgeValid = validateAge();
            const isPlanValid = validatePlan();

            const resultEl = document.getElementById('result');
            resultEl.style.display = 'block';

            if(isNameValid && isEmailValid && isMobileValid && isAgeValid && isPlanValid){
                resultEl.textContent = 'Gym Admission Successful!';
                resultEl.className = 'result-message result-success';
            } else {
                resultEl.textContent = 'Please Fill All Fields Correctly.';
                resultEl.className = 'result-message result-error';

                if(!inputs.name.value) showError('nameError', 'Name is required');
                if(!inputs.email.value) showError('emailError', 'Email is required');
                if(!inputs.mobile.value) showError('mobileError', 'Mobile is required');
                if(!inputs.age.value) showError('ageError', 'Age is required');
                if(!inputs.plan.value) showError('planError', 'Select a Membership Plan');
            }
        });
    </script>
</body>
</html>
```
